 
Команды:
 - вывод всех установленных проограмм:
  adb shell pm list packages -3

 - выдача прав на установку:
  adb shell appops set org.telegram.messenger.web REQUEST_INSTALL_PACKAGES allow

 - выдача доступа к файлам
  adb shell pm grant com.zeekrmate.app android.permission.READ_EXTERNAL_STORAGE
  adb shell pm grant com.zeekrmate.app android.permission.WRITE_EXTERNAL_STORAGE
  adb shell pm grant com.zeekrmate.app android.permission.READ_MEDIA_IMAGES
  adb shell pm grant com.zeekrmate.app android.permission.READ_MEDIA_VIDEO
  adb shell pm grant com.zeekrmate.app android.permission.READ_MEDIA_AUDIO
  adb shell pm grant com.zeekrmate.app android.permission.ACCESS_MEDIA_LOCATION
  adb shell pm grant com.zeekrmate.app android.permission.READ_MEDIA_VISUAL_USER_SELECTED

  adb shell appops set --uid com.zeekrmate.app MANAGE_EXTERNAL_STORAGE allow
  adb shell appops set com.zeekrmate.app MANAGE_EXTERNAL_STORAGE allow
  adb shell appops set com.zeekrmate.app WRITE_EXTERNAL_STORAGE allow
  adb shell appops set com.zeekrmate.app READ_EXTERNAL_STORAGE allow
  adb shell appops set com.zeekrmate.app LEGACY_STORAGE allow
  adb shell appops set com.zeekrmate.app NO_ISOLATED_STORAGE allow

  adb shell appops set com.zeekrmate.app AUTO_REVOKE_PERMISSIONS_IF_UNUSED ignore

  adb shell am force-stop com.zeekrmate.app
  adb shell monkey -p com.zeekrmate.app -c android.intent.category.LAUNCHER 1
  
  
  
как сделать управление яндексом
  запуск из powershell
  @'
  #!/system/bin/sh
  logcat -c
  logcat FDBusClient:I *:S | while IFS= read -r line
  do
    echo "$line" | grep -q "RECEIVE" || continue
    echo "$line" | grep -q "MEDIA_UPDATE_MEDIA_STATUS_REQUEST_CONTROL" || continue
    echo "$line" | grep -q '"dataType":5' && input keyevent 87
    echo "$line" | grep -q '"dataType":6' && input keyevent 88
    echo "$line" | grep -q '"dataType":7' && input keyevent 85
  done
  '@ -replace "`r`n","`n" | adb shell "cat > /data/local/tmp/ym-swc.sh; chmod 755 /data/local/tmp/ym-swc.sh"

  adb shell sh /data/local/tmp/ym-swc.sh

  adb shell "nohup sh /data/local/tmp/ym-swc.sh >/data/local/tmp/ym-swc.log 2>&1 &"
